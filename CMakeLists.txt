cmake_minimum_required(VERSION 3.16)
project(LikeMake VERSION 1.0.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(lm_common STATIC src/common.c)
target_include_directories(lm_common PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
if(MSVC)
  target_compile_options(lm_common PRIVATE /W4)
else()
  target_compile_options(lm_common PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(LikeMake src/likemake.c)
add_executable(LikeMakefileBuilder src/builder.c)
target_link_libraries(LikeMake PRIVATE lm_common)
target_link_libraries(LikeMakefileBuilder PRIVATE lm_common)
if(NOT MSVC)
  target_compile_options(LikeMake PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(LikeMakefileBuilder PRIVATE -Wall -Wextra -Wpedantic)
else()
  target_compile_options(LikeMake PRIVATE /W4)
  target_compile_options(LikeMakefileBuilder PRIVATE /W4)
endif()
install(TARGETS LikeMake LikeMakefileBuilder RUNTIME DESTINATION bin)
enable_testing()
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
  add_test(NAME likemake_integration COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_integration.py" "$<TARGET_FILE:LikeMake>" "$<TARGET_FILE:LikeMakefileBuilder>")
endif()
